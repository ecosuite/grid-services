cmake_minimum_required(VERSION 3.16)
project(grid-services CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if (NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif ()

option(GRID_SERVICES_BUILD_TESTS "Build the test suites" ON)
option(GRID_SERVICES_WERROR "Treat warnings in the project's own code as errors" OFF)

find_package(PkgConfig REQUIRED)
pkg_check_modules(PQXX REQUIRED IMPORTED_TARGET libpqxx)
pkg_check_modules(SODIUM REQUIRED IMPORTED_TARGET libsodium)
find_package(spdlog REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(CURL REQUIRED)
find_package(XercesC REQUIRED)
find_package(Threads REQUIRED)

# -----------------------------------------------------------------------------
# Vendored OpenADR 2.0b stack, under vendor/: EPRI liboadr and its
# CodeSynthesis XSD bindings. Built without warnings; it is third-party code.

file(GLOB_RECURSE OADRSD_SOURCES CONFIGURE_DEPENDS vendor/oadrsd/oadrsd/*.cxx)
# Shared, so the bindings' polymorphic type registrations (static
# initialisers) are never dropped by the linker.
add_library(oadrsd SHARED ${OADRSD_SOURCES})
target_include_directories(oadrsd SYSTEM PUBLIC vendor/oadrsd vendor/xsd-4.0.0)
target_link_libraries(oadrsd PUBLIC XercesC::XercesC)
target_compile_options(oadrsd PRIVATE -w)

file(GLOB_RECURSE OADR_SOURCES CONFIGURE_DEPENDS vendor/oadr/oadr/*.cpp)
add_library(oadr SHARED ${OADR_SOURCES})
target_include_directories(oadr SYSTEM PUBLIC vendor/oadr)
target_link_libraries(oadr PUBLIC oadrsd CURL::libcurl Threads::Threads)
target_compile_options(oadr PRIVATE -w)

# -----------------------------------------------------------------------------
# Vendored IEEE 2030.5-2018 bindings with the CSIP-AUS v1.1 extensions, under
# vendor/csipaussd, generated by its schema/generate.sh. Built without
# warnings; it is generated code.

file(GLOB CSIPAUSSD_SOURCES CONFIGURE_DEPENDS vendor/csipaussd/csipaussd/v11/*.cxx)
add_library(csipaussd STATIC ${CSIPAUSSD_SOURCES})
target_include_directories(csipaussd SYSTEM PUBLIC vendor/csipaussd vendor/xsd-4.0.0)
target_link_libraries(csipaussd PUBLIC XercesC::XercesC)
target_compile_options(csipaussd PRIVATE -w)

# -----------------------------------------------------------------------------
# grid-services
#
#   core             protocol-neutral types, adapter and store interfaces,
#                    runner
#   crypto           sealed secrets and client credentials
#   http             protocol-neutral HTTP client over libcurl
#   datum            datums and how a window of them becomes a reported value
#   solarnetwork     reads datums from SolarNetwork
#   store            Postgres implementation of the store interfaces
#   supervisor       reconciles runners against the database
#   openadr2b_model  OpenADR 2.0b's report types and store interface, which
#                    the store implements without liboadr
#   openadr2b        OpenADR 2.0b adapter
#   csipaus_model    CSIP-AUS's store interface
#   csipaus          CSIP-AUS (IEEE 2030.5-2018) adapter
#
# A protocol lives under src/protocols/<protocol>. core depends on no
# protocol, nor on Postgres.

set(GRID_SERVICES_WARNINGS -Wall -Wextra -Wno-missing-field-initializers)
if (GRID_SERVICES_WERROR)
    list(APPEND GRID_SERVICES_WARNINGS -Werror)
endif ()

add_library(grid_services_datum STATIC src/datum/datum.cpp)
target_include_directories(grid_services_datum PUBLIC src)
target_compile_options(grid_services_datum PRIVATE ${GRID_SERVICES_WARNINGS})

add_library(grid_services_core STATIC
        src/core/adapter.cpp
        src/core/event_sink.cpp
        src/core/runner.cpp
        src/core/telemetry.cpp
        src/core/types.cpp)
target_include_directories(grid_services_core PUBLIC src)
target_link_libraries(grid_services_core PUBLIC
        grid_services_datum nlohmann_json::nlohmann_json spdlog::spdlog Threads::Threads)
target_compile_options(grid_services_core PRIVATE ${GRID_SERVICES_WARNINGS})

add_library(grid_services_crypto STATIC
        src/crypto/client_credential.cpp
        src/crypto/sealed_box.cpp)
target_include_directories(grid_services_crypto PUBLIC src)
target_link_libraries(grid_services_crypto PUBLIC grid_services_core PkgConfig::SODIUM)
target_compile_options(grid_services_crypto PRIVATE ${GRID_SERVICES_WARNINGS})

add_library(grid_services_http STATIC
        src/http/curl_http_client.cpp
        src/http/http_client.cpp)
target_link_libraries(grid_services_http PUBLIC grid_services_core CURL::libcurl)
target_compile_options(grid_services_http PRIVATE ${GRID_SERVICES_WARNINGS})

add_library(grid_services_solarnetwork STATIC
        src/datum/snws2.cpp
        src/datum/solarnetwork_client.cpp)
target_link_libraries(grid_services_solarnetwork PUBLIC
        grid_services_datum nlohmann_json::nlohmann_json PkgConfig::SODIUM CURL::libcurl)
target_compile_options(grid_services_solarnetwork PRIVATE ${GRID_SERVICES_WARNINGS})

add_library(grid_services_openadr2b_model STATIC
        src/protocols/openadr2b/reports.cpp
        src/protocols/openadr2b/store.cpp)
target_link_libraries(grid_services_openadr2b_model PUBLIC grid_services_core)
target_compile_options(grid_services_openadr2b_model PRIVATE ${GRID_SERVICES_WARNINGS})

add_library(grid_services_csipaus_model STATIC src/protocols/csipaus/store.cpp)
target_link_libraries(grid_services_csipaus_model PUBLIC grid_services_core)
target_compile_options(grid_services_csipaus_model PRIVATE ${GRID_SERVICES_WARNINGS})

add_library(grid_services_store STATIC
        src/store/pg_csipaus_store.cpp
        src/store/pg_openadr2b_store.cpp
        src/store/pg_store.cpp)
target_link_libraries(grid_services_store PUBLIC
        grid_services_core grid_services_openadr2b_model grid_services_csipaus_model
        PkgConfig::PQXX)
target_compile_options(grid_services_store PRIVATE ${GRID_SERVICES_WARNINGS})

add_library(grid_services_supervisor STATIC src/supervisor/supervisor.cpp)
target_link_libraries(grid_services_supervisor PUBLIC grid_services_core)
target_compile_options(grid_services_supervisor PRIVATE ${GRID_SERVICES_WARNINGS})

add_library(grid_services_openadr2b STATIC
        src/protocols/openadr2b/curl_transport.cpp
        src/protocols/openadr2b/normalize.cpp
        src/protocols/openadr2b/openadr2b_adapter.cpp
        src/protocols/openadr2b/report_requests.cpp
        src/protocols/openadr2b/report_service.cpp
        src/protocols/openadr2b/tick_scheduler.cpp)
target_link_libraries(grid_services_openadr2b PUBLIC
        grid_services_core grid_services_openadr2b_model grid_services_crypto
        grid_services_http grid_services_solarnetwork oadr)
target_compile_options(grid_services_openadr2b PRIVATE ${GRID_SERVICES_WARNINGS})

add_library(grid_services_csipaus STATIC
        src/protocols/csipaus/controls.cpp
        src/protocols/csipaus/csipaus_adapter.cpp
        src/protocols/csipaus/documents.cpp
        src/protocols/csipaus/identity.cpp
        src/protocols/csipaus/limits.cpp
        src/protocols/csipaus/reporter.cpp
        src/protocols/csipaus/reporting.cpp
        src/protocols/csipaus/resource_schedule.cpp
        src/protocols/csipaus/sep_client.cpp)
target_link_libraries(grid_services_csipaus PUBLIC
        grid_services_core grid_services_csipaus_model grid_services_crypto
        grid_services_http grid_services_solarnetwork csipaussd)
target_compile_options(grid_services_csipaus PRIVATE ${GRID_SERVICES_WARNINGS})

add_executable(grid-services src/main.cpp)
target_link_libraries(grid-services PRIVATE
        grid_services_supervisor grid_services_store grid_services_openadr2b
        grid_services_csipaus grid_services_crypto)
target_compile_options(grid-services PRIVATE ${GRID_SERVICES_WARNINGS})

# -----------------------------------------------------------------------------
# Tests

if (GRID_SERVICES_BUILD_TESTS)
    enable_testing()
    find_package(GTest REQUIRED)

    # liboadr's own suite. It loads fixtures from xml/ under the build
    # directory.
    file(GLOB_RECURSE LIBOADR_TEST_SOURCES CONFIGURE_DEPENDS
            vendor/oadrtest/oadrtest/*.cpp)
    add_executable(liboadr_tests ${LIBOADR_TEST_SOURCES})
    target_link_libraries(liboadr_tests PRIVATE oadr GTest::gtest)
    target_compile_options(liboadr_tests PRIVATE -w)
    file(COPY vendor/oadrtest/oadrtest DESTINATION ${CMAKE_BINARY_DIR}/xml
            FILES_MATCHING PATTERN "*.xml" PATTERN "*.txt")
    add_test(NAME liboadr COMMAND liboadr_tests
            WORKING_DIRECTORY ${CMAKE_BINARY_DIR})

    # The CSIP-AUS bindings' suite. Its fixtures are read from the source
    # tree.
    file(GLOB_RECURSE CSIPAUS_TEST_SOURCES CONFIGURE_DEPENDS
            vendor/csipaustest/csipaustest/*.cpp)
    add_executable(csipaus_tests ${CSIPAUS_TEST_SOURCES})
    target_include_directories(csipaus_tests PRIVATE vendor/csipaustest/csipaustest)
    target_link_libraries(csipaus_tests PRIVATE csipaussd GTest::gtest)
    target_compile_definitions(csipaus_tests PRIVATE
            CSIPAUS_FIXTURES_DIR="${PROJECT_SOURCE_DIR}/vendor/csipaustest/csipaustest/fixtures")
    target_compile_options(csipaus_tests PRIVATE ${GRID_SERVICES_WARNINGS})
    add_test(NAME csipaus COMMAND csipaus_tests)

    add_subdirectory(tests)
endif ()
